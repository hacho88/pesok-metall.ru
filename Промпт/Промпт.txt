 ШАГ 2: NEXT.JS МУЛЬТИПОДДОМЕННЫЙ MIDDLEWAREЗапрограммируй логику роутинга в файле src/middleware.ts. Middleware должен перехватывать запросы со сторонних субдоменов (например, podolsk.pesok-metall.ru) и незаметно подменять их внутри директории _geo/[slug] без изменения URL-адреса у пользователя.
import { NextResponse } from 'next/server'
import type { NextRequest } from 'next/server'

export function middleware(request: NextRequest) {
  const url = request.nextUrl
  const hostname = request.headers.get('host') || ''
  
  // Пропускаем системные файлы, API и статику
  if (
    url.pathname.startsWith('/_next') ||
    url.pathname.startsWith('/api') ||
    url.pathname.includes('.')
  ) {
    return NextResponse.next()
  }

  const hostnameParts = hostname.split('.')
  
  // Локально: balashiha.localhost:3000 | В проде: balashiha.pesok-metall.ru
  if (hostnameParts.length >= 3) {
    const subdomain = hostnameParts[0].toLowerCase()
    
    if (subdomain !== 'www' && subdomain !== 'pesok-metall') {
      // Магия рерайта: незаметно подменяем страницу под гео-зону
      url.pathname = `/_geo/${subdomain}${url.pathname}`
      return NextResponse.rewrite(url)
    }
  }
  
  return NextResponse.next()
}

 ШАГ 4: ИНТЕРАКТИВНЫЙ ИИ-КАЛЬКУЛЯТОР И ЛОГИКА АВТОПАРКАЗапрограммируй математическую и логическую модель калькулятора доставки на фронтенде (src/components/AiCalculator.tsx). Калькулятор должен на лету переводить объем сыпучих материалов в мешки по 30 кг / Биг-Беги и выбирать оптимальную машину из базы данных.
// Алгоритм перевода объема в физическую тару на основе плотности
export function calculateBulkMaterials(areaM2: number, thicknessCm: number, density: number) {
  const volumeM3 = areaM2 * (thicknessCm / 100);
  const totalWeightKg = volumeM3 * density;
  
  const bags30kg = Math.ceil(totalWeightKg / 30);
  const bigBags1Ton = Math.ceil(totalWeightKg / 1000);
  
  return { volumeM3, totalWeightKg, bags30kg, bigBags1Ton };
}

// Алгоритм автоподбора оптимального грузовика под габариты и вес заказа
export function assignOptimalVehicle(weightKg: number, maxLengthMeters: number, fleet: any[]) {
  // Фильтруем только активные машины, которые проходят по весу и длине металла
  const fitVehicles = fleet
    .filter(v => v.isActive && weightKg <= v.maxWeightKg && maxLengthMeters <= v.maxLengthMeters)
    .sort((a, b) => Number(a.baseFare) - Number(b.baseFare)); // Сортируем от самых дешевых по подаче
    
  return fitVehicles.length > 0 ? fitVehicles[0] : null; // Возвращаем идеальный транспорт
}
